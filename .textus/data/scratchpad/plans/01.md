# Session/API Extraction Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use subagent-driven-development or executing-plans.

**Goal:** Extract LoopNavigator from Session, create Textus::API dispatch engine, and make Session a value object used by API.

**Architecture:** Three-layer split: (1) `Textus::API` is the public entry point with explicit action methods (`get`, `put`, etc.) and session lifecycle (`open_session`, `resume_session`). (2) `Textus::API::Session` is a value object (role, session_id, payload) used as context for dispatch. (3) Internal components (`Loop`, `SessionStore`, `Workspace`) are composed by API. All three surfaces (CLI, Web, MCP) call API directly; Gate becomes an internal detail.

**Tech Stack:** Ruby 3.3+, Sequel, SQLite, RSpec

- [ ] **Step 1: Extract LoopNavigator state machine**

```ruby
require_relative "loop/transitions"

module Textus
  class API
    class Loop
      NodeEntry = Data.define(:path, :entered_at, :exited_at) do
        def node   = path.first
        def topic  = path[1]
        def item   = path[2]
        def current? = exited_at.nil?
        def duration  = (exited_at || Time.now.utc) - entered_at
      end

      def initialize(node_names, store:, session:)
        @node_names = node_names
        @store = store
        @session = session
        @stack = restore_stack(session.payload)
      end

      def enter(path)
        segments = path.map { |s| s.is_a?(String) && !s.match?(/\A[0-9]|-/) ? s.to_sym : s }
        Loop::Transitions.validate!(current_node&.path, segments, node_names: @node_names)

        current = current_node
        changing_node = current&.current? && current.node != segments.first
        fire_node_exit(current) if changing_node

        entry = NodeEntry.new(path: segments, entered_at: Time.now.utc, exited_at: nil)
        @stack.push(entry)

        fire_node_enter(entry) if changing_node || @stack.size == 1
        persist
        self
      end

      def exit
        entry = current_node
        raise UsageError.new("no node to exit") unless entry

        to_remove = @stack.reverse_each.with_object([]) do |e, acc|
          break acc if e.node != entry.node
          acc << e
        end

        to_remove.each { |e| @stack.delete(e) }
        fire_node_enter(current_node) if current_node
        persist
      end

      def back
        entry = current_node
        raise UsageError.new("no node to step back from") unless entry

        @stack.delete(entry)
        fire_node_enter(current_node) if current_node&.current?
        persist
      end

      def switch(path)
        exit
        enter(path)
      end

      def first
        enter([@node_names.first])
      end

      def status
        { current_node: current_node&.node&.to_s,
          current_topic: current_node&.topic&.to_s,
          current_item: current_node&.item&.to_s,
          stack: summary, available_nodes: @node_names.map(&:to_s) }
      end

      def current
        current_node
      end

      def to_h
        { stack: @stack.map { |e| e.path.map(&:to_s) } }
      end

      private

      def current_node
        @stack.reverse_each.find(&:current?) || @stack.last
      end

      def summary
        @stack.map do |e|
          { path: e.path.map(&:to_s), entered_at: e.entered_at.iso8601,
            exited_at: e.exited_at&.iso8601, current: e.current?, duration_s: e.duration }
        end
      end

      def restore_stack(payload)
        (payload["stack"] || []).map do |path|
          NodeEntry.new(path: path.map(&:to_sym), entered_at: Time.now.utc, exited_at: nil)
        end
      end

      def persist
        @store.update_payload(@session.id, to_h)
      end

      def fire_node_enter(entry)
        dispatch(:node_entered, path: entry.path.map(&:to_s))
      rescue StandardError
        nil
      end

      def fire_node_exit(entry)
        entry.exited_at = Time.now.utc
        dispatch(:node_exited, path: entry.path.map(&:to_s))
      rescue StandardError
        nil
      end

      def dispatch(action, **params)
        request = Adapters::Request.new(action:, params:, role: @session.role, session_id: @session.id)
        response = @gate.dispatch(request)
        raise response.error if response.error
        response.payload
      end
    end
  end
end
```

```ruby
module Textus
  class API
    class Workspace
      def initialize(store_engine)
        @store = store_engine
      end

      def create(session_id, role:)
        return unless role == "agent"

        key = "scratchpad.sessions.#{session_id}"
        body = JSON.generate({
          session_id:, role:, status: "active",
          created_at: Time.now.utc.iso8601,
        })
        @store.put(key:, meta: {}, body:, content: nil)
      end

      def close(session_id, loop_status: nil, nodes_checked: nil)
        key = "scratchpad.sessions.#{session_id}"
        env = @store.get(key:)
        return unless env

        current = JSON.parse(env.body || "{}")
        feedback = { node: "feedback", confidence: "medium",
                     signal: "session_closed",
                     summary: "Session #{session_id[0..8]} sealed." }
        feedback[:loop] = loop_status if loop_status
        feedback[:nodes_checked] = nodes_checked if nodes_checked

        content = current.merge("status" => "sealed", "closed_at" => Time.now.utc.iso8601,
                                "feedback" => feedback)
        @store.put(key:, meta: {}, body: JSON.generate(content), content: nil)
      end
    end
  end
end
```

- [ ] **Step 1: Create API class with all action methods**

```ruby
module Textus
  class API
    def initialize(gate:, store:)
      @gate = gate
      @store = store
      @session_store = Protocol::SessionStore.new(@store.db)
      @workspace = Workspace.new(@store)
    end

    # ── Session lifecycle ──

    def open_session(role:)
      sid = @session_store.create(
        id: "#{Time.now.utc.strftime("%Y-%m-%d")}-#{SecureRandom.hex(4)}",
        role:, surface_id: nil,
      )[:id]
      @workspace.create(sid, role:)
      session = Session.new(id: sid, role:)
      if role == "agent"
        loop = Loop.new(loop_node_names, store: @session_store, session:)
        loop.first
      end
      session
    end

    def resume_session(id:)
      row = @session_store.find(id)
      raise UsageError.new("session not found: #{id}") unless row
      Session.new(id: row[:id], role: row[:role],
                  correlation_id: row[:correlation_id], payload: row[:payload])
    end

    def close_session(session, nodes_checked: nil)
      @session_store.seal(session.id)
      @workspace.close(session.id, loop_status: session.payload["loop"], nodes_checked:) if session.agent?
    end

    # ── Loop ──

    def loop_for(session)
      Loop.new(loop_node_names, store: @session_store, session:)
    end

    # ── Dispatch ──

    def get(key:, session:)          = dispatch(:get, session:, key:)
    def put(key:, data:, session:, **opts) = dispatch(:put, session:, key:, data:, **opts)
    def list(session:, **opts)        = dispatch(:list, session:, **opts)
    def propose(key:, data:, session:, **opts) = dispatch(:propose, session:, key:, data:, **opts)
    def accept(pending_key:, session:) = dispatch(:accept, session:, pending_key:)
    def reject(pending_key:, session:, **opts) = dispatch(:reject, session:, pending_key:, **opts)
    def audit(session:, **filters)    = dispatch(:audit, session:, **filters)
    def blame(key:, session:, **opts) = dispatch(:blame, session:, key:, **opts)
    def deps(key:, session:)          = dispatch(:deps, session:, key:)
    def graph(key:, session:, **opts)  = dispatch(:graph, session:, key:, **opts)
    def uid(key:, session:)           = dispatch(:uid, session:, key:)
    def where(key:, session:)         = dispatch(:where, session:, key:)
    def rdeps(key:, session:)         = dispatch(:rdeps, session:, key:)
    def pulse(session:, **opts)       = dispatch(:pulse, session:, **opts)
    def drain(session:, **opts)       = dispatch(:drain, session:, **opts)
    def schema_list(session:)         = dispatch(:schema_list, session:)
    def schema_show(key:, session:)   = dispatch(:schema_show, session:, key:)
    def key_delete(key:, session:, **opts) = dispatch(:key_delete, session:, key:, **opts)
    def key_mv(from:, to:, session:, **opts) = dispatch(:key_mv, session:, from:, to:, **opts)
    def boot                          = Protocol::Boot.build(container: @store.container)
    def cursor                        = @store.audit_log.latest[:seq]

    private

    def dispatch(action, session:, **params)
      request = Adapters::Request.new(
        action:, params:, role: session.role,
        session_id: session.id, correlation_id: session.correlation_id,
      )
      response = @gate.dispatch(request)
      raise response.error if response.error
      response.payload
    end

    def loop_node_names
      @store.manifest.data.entries
        .select { |e| e.key.start_with?("knowledge.loop.") && e.key.count(".") == 2 }
        .map { |e| e.key.split(".").last.to_sym }
    end
  end
end
```
